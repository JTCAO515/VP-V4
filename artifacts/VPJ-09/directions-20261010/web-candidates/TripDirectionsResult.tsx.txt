"use client";
import { useEffect, useRef, useState } from 'react';
import { createPasswordAuthClient } from '@/lib/server/identity/browser-auth-client';
import { parseResultArtifactReadV2, type ResultArtifactReadV2 } from '@/lib/server/artifacts/result-v2-contract';
import { directionsReceipt, object } from '@/lib/server/planning/directions/protocol';
import type { TravelDirectionsContent } from '@/lib/server/planning/directions/contract';
import type { PendingProposalRead } from '@/lib/server/identity/user-data-adapter';
import type { Locale } from '@/lib/i18n';
import styles from './TripDirectionsResult.module.css';
type Read=ResultArtifactReadV2&{content:TravelDirectionsContent};
type Edit={ordinal:number;destination:string;activities:string;baseRevision:number};
export function TripDirectionsResult({tripId,tripVersion,locale,bindBlocked,onProposalCreated}:{tripId:string;tripVersion:number;locale:Locale;bindBlocked:boolean;onProposalCreated:(p:PendingProposalRead)=>void}){
 const zh=locale==='zh',generation=useRef(0);
 const [read,setRead]=useState<Read|null>(null),[state,setState]=useState<'loading'|'ready'|'empty'|'unavailable'>('loading');
 const [refresh,setRefresh]=useState(0),[busy,setBusy]=useState(false),[edit,setEdit]=useState<Edit|null>(null),[startDate,setStartDate]=useState(''),[notice,setNotice]=useState('');
 const pending=useRef<{action:string;bytes:string}|null>(null);
 useEffect(()=>{
  const reload=()=>{generation.current++;setRead(null);setEdit(null);setBusy(false);pending.current=null;setRefresh(n=>n+1);};
  document.addEventListener('visibilitychange',reload);window.addEventListener('pageshow',reload);window.addEventListener('online',reload);
  const auth=createPasswordAuthClient()?.auth.onAuthStateChange(reload);
  return()=>{document.removeEventListener('visibilitychange',reload);window.removeEventListener('pageshow',reload);window.removeEventListener('online',reload);auth?.data.subscription.unsubscribe();};
 },[]);
 useEffect(()=>{
  const current=++generation.current,controller=new AbortController();setRead(null);setState('loading');
  if(document.visibilityState==='hidden')return()=>controller.abort();
  void (async()=>{try{
   const response=await fetch(`/api/trips/${tripId}/directions`,{cache:'no-store',signal:controller.signal});const raw:unknown=await response.json();
   if(current!==generation.current)return;
   if(!response.ok||!object(raw)||raw.version!==2){setState('unavailable');return;}
   if(object(raw.data)&&raw.data.kind==='empty'){setState('empty');return;}
   const r=parseResultArtifactReadV2(raw.data);
   if(!r||r.content.schemaVersion!=='travel-directions/1'||r.source.tripId!==tripId){setState('unavailable');return;}
   setRead({...r,content:r.content});setState('ready');
  }catch{if(current===generation.current&&!controller.signal.aborted)setState('unavailable');}})();
  return()=>{generation.current++;controller.abort();};
 },[tripId,tripVersion,refresh]);
 const canAct=!!read&&read.current&&read.lifecycle==='active'&&read.source.tripVersion===tripVersion&&!busy;
 async function act(action:'choose'|'save'|'edit'|'bind',extra:Record<string,unknown>={},retry=false){
  if(!read||!canAct||action==='bind'&&bindBlocked)return;
  const current=generation.current;
  const params={artifactId:read.artifactId,expectedRevision:action==='edit'&&edit?edit.baseRevision:read.revision,operationId:crypto.randomUUID(),...extra};
  const bytes=retry&&pending.current?.action===action?pending.current.bytes:JSON.stringify(params);
  pending.current={action,bytes};setBusy(true);setNotice('');
  try{
   const response=await fetch(`/api/trips/${tripId}/directions/${action}`,{method:'POST',headers:{'content-type':'application/json'},body:bytes});
   const receipt:unknown=await response.json();if(current!==generation.current)return;
   if(!response.ok||!object(receipt)||receipt.version!==1||!directionsReceipt(action,Object.fromEntries(Object.entries(receipt).filter(([k])=>k!=='version')),JSON.parse(bytes))){setNotice(zh?'当前资料已变化或暂不可用，请刷新后重试。':'Source changed or unavailable. Refresh before trying again.');pending.current=null;return;}
   if(action==='bind'){
    const response=await fetch(`/api/trips/${tripId}/proposal?proposalId=${receipt.proposalId}`,{cache:'no-store'});const p=await response.json() as PendingProposalRead;
    if(current!==generation.current)return;
    if(!response.ok||p.trip?.id!==tripId||p.trip.headVersion!==tripVersion||p.proposal?.id!==receipt.proposalId||p.proposal.revision!==receipt.proposalRevision||p.proposal.baseTripVersion!==tripVersion||p.proposal.status!=='pending'||p.proposal.stale||typeof p.proposal.digest!=='string'||!/^[a-f0-9]{64}$/.test(p.proposal.digest))throw Error('original proposal unavailable');
    onProposalCreated(p);setNotice(zh?'已生成原计划差异，请在下方审阅后明确确认。':'The original Trip diff is ready below for your explicit confirmation.');
   }else setNotice(zh?'已保存新版本。':'New revision saved.');
   pending.current=null;setEdit(null);setRefresh(n=>n+1);
  }catch{if(current===generation.current)setNotice(zh?'结果未知，可用相同请求重试；不要重复新建。':'Outcome unknown. Retry the same request.');}
  finally{if(current===generation.current)setBusy(false);}
 }
 if(state==='empty')return null;
 return <section className={styles.card} aria-label={zh?'旅行方向':'Travel directions'} data-testid="trip-directions">
  <div className={styles.header}><h2>{zh?'旅行方向与相对日草稿':'Directions and relative-day draft'}</h2><button type="button" disabled={busy} onClick={()=>{pending.current=null;setEdit(null);setRefresh(n=>n+1);}}>{zh?'刷新':'Refresh'}</button></div>
  <p role="status">{state==='loading'?(zh?'正在读取原成果…':'Reading the original result…'):state==='unavailable'?(zh?'目前无法读取合资格的方向成果。':'A qualified directions result is unavailable.'):notice}</p>
  {read?<><h3>{read.content.title}</h3><p>{read.content.summary}</p><p>{zh?'日期：':'Dates: '}{read.content.intake.dates?`${read.content.intake.dates.startDate} – ${read.content.intake.dates.endDate}`:zh?'未定，可稍后补':'Unknown; add later'} · {zh?'预算：':'Budget: '}{read.content.intake.budget?`${read.content.intake.budget.currency} ${read.content.intake.budget.totalMinorUnits/100}`:zh?'未知，可跳过':'Unknown; can skip'}</p>
   {!read.current?<p>{zh?'这是原有的历史版本；当前来源已变化，不能继续修改或提交。':'Historical revision: its source changed, so actions are unavailable.'}</p>:null}
   <div className={styles.options}>{read.content.directions.map(d=><article key={d.id}><h4>{d.title}</h4><p>{d.tradeoff}</p><button type="button" disabled={!canAct||read.content.selectedDirectionId===d.id} onClick={()=>void act('choose',{directionId:d.id})}>{read.content.selectedDirectionId===d.id?(zh?'已选择':'Selected'):(zh?'选这个方向':'Choose direction')}</button></article>)}</div>
   <button type="button" disabled={!canAct||read.content.selectedDirectionId===null} onClick={()=>void act('save')}>{zh?'保存相对日草稿':'Save relative draft'}</button>
   {read.content.draft?<><ul>{read.content.draft.limitations.map(l=><li key={l}>{l}</li>)}</ul><ol className={styles.days}>{read.content.draft.days.map(d=><li key={d.ordinal}><h4>{zh?'第':'Day '}{d.ordinal}{zh?'天':''} · {d.destination}</h4><ul>{d.activities.map(a=><li key={a}>{a}</li>)}</ul><button type="button" disabled={!canAct} onClick={()=>setEdit({ordinal:d.ordinal,destination:d.destination,activities:d.activities.join('\n'),baseRevision:read.revision})}>{zh?'修改这一天':'Edit this day'}</button></li>)}</ol>
    {edit?<form onSubmit={e=>{e.preventDefault();void act('edit',{replacements:[{ordinal:edit.ordinal,destination:edit.destination.trim(),activities:edit.activities.split('\n').map(a=>a.trim()).filter(Boolean)}]});}}><label>{zh?'目的地':'Destination'}<input maxLength={80} value={edit.destination} onChange={e=>setEdit({...edit,destination:e.target.value})}/></label><label>{zh?'主题，每行一个':'One activity theme per line'}<textarea maxLength={1288} value={edit.activities} onChange={e=>setEdit({...edit,activities:e.target.value})}/></label><button type="submit" disabled={!canAct||edit.baseRevision!==read.revision}>{zh?'保存本日修改':'Save this day'}</button><button type="button" onClick={()=>setEdit(null)}>{zh?'取消修改':'Cancel edit'}</button></form>:null}
    <form onSubmit={e=>{e.preventDefault();void act('bind',{tripId,expectedTripVersion:tripVersion,startDate});}}><label>{zh?'明确开始日期后，生成同 Trip 差异':'Choose a start date to prepare the same Trip diff'}<input type="date" required value={startDate} onChange={e=>setStartDate(e.target.value)}/></label><button type="submit" disabled={!canAct||bindBlocked||read.content.draft.days.length===0}>{zh?'生成待审阅计划':'Prepare plan for review'}</button></form>
   </>:null}
   {pending.current?<button type="button" disabled={!canAct} onClick={()=>void act(pending.current!.action as 'choose'|'save'|'edit'|'bind',{},true)}>{zh?'重试相同请求':'Retry same request'}</button>:null}
  </>:null}
 </section>;
}

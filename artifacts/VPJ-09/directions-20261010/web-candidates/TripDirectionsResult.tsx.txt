"use client";
import { useEffect, useRef, useState } from 'react';
import { createPasswordAuthClient } from '@/lib/server/identity/browser-auth-client';
import { parseResultArtifactReadV2, type ResultArtifactReadV2 } from '@/lib/server/artifacts/result-v2-contract';
import { directionsReceipt, object } from '@/lib/server/planning/directions/protocol';
import { directionsClientActor, directionsClientScope, directionsClientSourceKey, directionsClientLeaseCurrent, directionsClientPendingCurrent, directionsClientActorChanged, type DirectionsClientPending } from '@/lib/server/planning/directions/client-state';
import type { TravelDirectionsContent } from '@/lib/server/planning/directions/contract';
import type { PendingProposalRead } from '@/lib/server/identity/user-data-adapter';
import type { Locale } from '@/lib/i18n';
import styles from './TripDirectionsResult.module.css';
type Read=ResultArtifactReadV2&{content:TravelDirectionsContent};
type Cache={scopeKey:string;record:Read;startedAt:number;deadline:number};
type Edit={scopeKey:string;sourceKey:string;artifactId:string;ordinal:number;destination:string;activities:string;baseRevision:number};
export function TripDirectionsResult({tripId,tripVersion,locale,bindBlocked,onProposalCreated}:{tripId:string;tripVersion:number;locale:Locale;bindBlocked:boolean;onProposalCreated:(p:PendingProposalRead)=>void}){
 const zh=locale==='zh',generation=useRef(0),actorRef=useRef<string|null>(null);
 const [actorKey,setActorKey]=useState<string|null>(null),[cache,setCache]=useState<Cache|null>(null),[state,setState]=useState<'loading'|'ready'|'empty'|'unavailable'|'expired'>('loading');
 const [refresh,setRefresh]=useState(0),[busy,setBusy]=useState(false),[edit,setEdit]=useState<Edit|null>(null),[startDate,setStartDate]=useState(''),[notice,setNotice]=useState('');
 const pending=useRef<DirectionsClientPending|null>(null),cacheRef=useRef<Cache|null>(null),write=useRef<AbortController|null>(null);
 const scopeKey=directionsClientScope(actorKey,tripId),context=useRef({scopeKey,tripId,tripVersion,onProposalCreated});
 // Updated during render: callbacks cannot use the effect-cleanup gap to enter a new Trip.
 context.current={scopeKey,tripId,tripVersion,onProposalCreated};
 const publishCache=(next:Cache|null)=>{cacheRef.current=next;setCache(next);};
 const read=directionsClientLeaseCurrent(cache,scopeKey,performance.now())?cache!.record:null;
 useEffect(()=>{
  let active=true,checking=false,queued=false,authEpoch=0;const client=createPasswordAuthClient();
  const suspend=()=>{generation.current++;write.current?.abort();publishCache(null);setBusy(false);setRefresh(n=>n+1);};
  const verify=async()=>{
   if(!active)return;if(checking){queued=true;return;}checking=true;const capturedEpoch=authEpoch;
   try{
    const r=await client?.auth.getClaims();if(!active)return;if(capturedEpoch!==authEpoch){queued=true;return;}
    const next=r&&!r.error?directionsClientActor(r.data?.claims?.sub,r.data?.claims?.session_id):null;
    if(next===null){setActorKey(null);suspend();return;}
    if(directionsClientActorChanged(actorRef.current,next)){pending.current=null;setEdit(null);setStartDate('');}
    actorRef.current=next;setActorKey(next);suspend();
   }catch{if(active){setActorKey(null);suspend();}}finally{checking=false;if(active&&queued){queued=false;queueMicrotask(()=>void verify());}}
  };
  const resume=()=>{authEpoch++;suspend();queueMicrotask(()=>void verify());};
  document.addEventListener('visibilitychange',resume);window.addEventListener('pageshow',resume);window.addEventListener('online',resume);
  const auth=client?.auth.onAuthStateChange((event,session)=>{
   authEpoch++;suspend();
   if(event==='SIGNED_OUT'||session===null){actorRef.current=null;pending.current=null;setActorKey(null);setEdit(null);setStartDate('');}
   else queueMicrotask(()=>void verify()); // No Auth calls inside its callback lock.
  });
  void verify();
  return()=>{active=false;authEpoch++;generation.current++;write.current?.abort();document.removeEventListener('visibilitychange',resume);window.removeEventListener('pageshow',resume);window.removeEventListener('online',resume);auth?.data.subscription.unsubscribe();};
 },[]);
 useEffect(()=>{
  const current=++generation.current,controller=new AbortController(),startedAt=performance.now();publishCache(null);setState('loading');setBusy(false);write.current?.abort();
  if(!scopeKey){setState('unavailable');return()=>controller.abort();}
  if(document.visibilityState==='hidden')return()=>controller.abort();
  const live=()=>current===generation.current&&context.current.scopeKey===scopeKey&&context.current.tripId===tripId&&context.current.tripVersion===tripVersion;
  const timer=setTimeout(()=>{if(live()&&cacheRef.current?.startedAt===startedAt){publishCache(null);setState('expired');}controller.abort();},30000);
  void (async()=>{try{
   const response=await fetch(`/api/trips/${tripId}/directions`,{cache:'no-store',signal:controller.signal});const raw:unknown=await response.json();
   if(!live())return;if(performance.now()>=startedAt+30000){setState('expired');return;}
   if(!response.ok||!object(raw)||raw.version!==2){setState('unavailable');return;}
   if(object(raw.data)&&raw.data.kind==='empty'){setState('empty');return;}
   const r=parseResultArtifactReadV2(raw.data);
   if(!r||r.content.schemaVersion!=='travel-directions/1'||r.source.tripId!==tripId){setState('unavailable');return;}
   publishCache({scopeKey,record:{...r,content:r.content},startedAt,deadline:startedAt+30000});setState('ready');
  }catch{if(live())setState(controller.signal.aborted?'expired':'unavailable');}})();
  return()=>{generation.current++;controller.abort();clearTimeout(timer);};
 },[scopeKey,tripId,tripVersion,refresh]);
 const sourceKey=read?directionsClientSourceKey(read.source,read.basis.memories):null;
 const pendingHere=directionsClientPendingCurrent(pending.current,scopeKey,sourceKey);
 const baseCanAct=!!read&&read.current&&read.lifecycle==='active'&&read.source.tripId===tripId&&read.source.tripVersion===tripVersion&&!busy;
 const canAct=baseCanAct&&pending.current===null;
 async function act(action:'choose'|'save'|'edit'|'bind',extra:Record<string,unknown>={},retry=false){
  if(!read||!scopeKey||!sourceKey||!actorKey||!baseCanAct||(!retry&&pending.current!==null)||(retry&&!pendingHere)||action==='bind'&&bindBlocked)return;
  const current=generation.current,captured={scopeKey,tripId,tripVersion},startedAt=performance.now(),controller=new AbortController();write.current=controller;
  if(action==='edit'&&(!edit||edit.scopeKey!==scopeKey||edit.sourceKey!==sourceKey||edit.artifactId!==read.artifactId))return;
  const params={artifactId:read.artifactId,expectedRevision:action==='edit'&&edit?edit.baseRevision:read.revision,operationId:crypto.randomUUID(),...extra};
  if(!retry)pending.current={scopeKey,sourceKey,actorKey,tripId,artifactId:read.artifactId,revision:Number(params.expectedRevision),action,bytes:JSON.stringify(params)};
  const original=pending.current;if(!original||!directionsClientPendingCurrent(original,scopeKey,sourceKey))return;
  const live=()=>current===generation.current&&context.current.scopeKey===captured.scopeKey&&context.current.tripId===captured.tripId&&context.current.tripVersion===captured.tripVersion&&performance.now()<startedAt+30000;
  const timer=setTimeout(()=>controller.abort(),30000);setBusy(true);setNotice('');
  try{
   const response=await fetch(`/api/trips/${original.tripId}/directions/${original.action}`,{method:'POST',headers:{'content-type':'application/json'},body:original.bytes,signal:controller.signal});
   const receipt:unknown=await response.json();if(!live())return;
   if(!response.ok||!object(receipt)||receipt.version!==1||!directionsReceipt(original.action,Object.fromEntries(Object.entries(receipt).filter(([k])=>k!=='version')),JSON.parse(original.bytes))){setNotice(zh?'当前资料已变化或暂不可用，请刷新后重试。':'Source changed or unavailable. Refresh before trying again.');pending.current=null;return;}
   if(original.action==='bind'){
    const response=await fetch(`/api/trips/${original.tripId}/proposal?proposalId=${receipt.proposalId}`,{cache:'no-store',signal:controller.signal});const p=await response.json() as PendingProposalRead;
    if(!live())return;
    if(!response.ok||p.trip?.id!==captured.tripId||p.trip.headVersion!==captured.tripVersion||p.proposal?.id!==receipt.proposalId||p.proposal.revision!==receipt.proposalRevision||p.proposal.baseTripVersion!==captured.tripVersion||p.proposal.status!=='pending'||p.proposal.stale||typeof p.proposal.digest!=='string'||!/^[a-f0-9]{64}$/.test(p.proposal.digest))throw Error('original proposal unavailable');
    if(!live())return;context.current.onProposalCreated(p);setNotice(zh?'已生成原计划差异，请在下方审阅后明确确认。':'The original Trip diff is ready below for your explicit confirmation.');
   }else setNotice(zh?'已保存新版本，请重新读取原成果。':'New revision saved; reread the original result.');
   if(!live())return;pending.current=null;setEdit(null);setRefresh(n=>n+1);
  }catch{if(current===generation.current&&context.current.scopeKey===captured.scopeKey)setNotice(zh?'结果未知，可用相同请求重试；不要重复新建。':'Outcome unknown. Retry the same request.');}
  finally{clearTimeout(timer);if(write.current===controller)write.current=null;if(current===generation.current&&context.current.scopeKey===captured.scopeKey)setBusy(false);}
 }
 if(state==='empty')return null;
 return <section className={styles.card} aria-label={zh?'旅行方向':'Travel directions'} data-testid="trip-directions">
  <div className={styles.header}><h2>{zh?'旅行方向与相对日草稿':'Directions and relative-day draft'}</h2><button type="button" disabled={busy} onClick={()=>{setRefresh(n=>n+1);}}>{zh?'刷新':'Refresh'}</button></div>
  <p role="status">{state==='loading'?(zh?'正在读取原成果…':'Reading the original result…'):state==='unavailable'?(zh?'目前无法读取合资格的方向成果。':'A qualified directions result is unavailable.'):state==='expired'?(zh?'显示依据已到期，请重新读取。':'The display basis expired. Please reread.'):notice}</p>
  {pending.current&&!pendingHere&&pending.current.actorKey===actorKey?<p>{zh?'原上下文有未确认请求，请返回原 Trip 与成果重新读取并明确重试。':'The original context has an unresolved request. Return to its Trip and result to reread and explicitly retry.'}</p>:null}
  {pending.current&&!pendingHere&&pending.current.actorKey===actorKey?<button type="button" disabled={busy} onClick={()=>{pending.current=null;setEdit(null);setRefresh(n=>n+1);}}>{zh?'清除旧上下文未确认请求，重新读取当前成果':'Clear the old unresolved request and reread the current result'}</button>:null}
  {read?<><h3>{read.content.title}</h3><p>{read.content.summary}</p><p>{zh?'日期：':'Dates: '}{read.content.intake.dates?`${read.content.intake.dates.startDate} – ${read.content.intake.dates.endDate}`:zh?'未定，可稍后补':'Unknown; add later'} · {zh?'预算：':'Budget: '}{read.content.intake.budget?`${read.content.intake.budget.currency} ${read.content.intake.budget.totalMinorUnits/100}`:zh?'未知，可跳过':'Unknown; can skip'}</p>
   {!read.current?<p>{zh?'这是原有的历史版本；当前来源已变化，不能继续修改或提交。':'Historical revision: its source changed, so actions are unavailable.'}</p>:null}
   <div className={styles.options}>{read.content.directions.map(d=><article key={d.id}><h4>{d.title}</h4><p>{d.tradeoff}</p><button type="button" disabled={!canAct||read.content.selectedDirectionId===d.id} onClick={()=>void act('choose',{directionId:d.id})}>{read.content.selectedDirectionId===d.id?(zh?'已选择':'Selected'):(zh?'选这个方向':'Choose direction')}</button></article>)}</div>
   <button type="button" disabled={!canAct||read.content.selectedDirectionId===null} onClick={()=>void act('save')}>{zh?'保存相对日草稿':'Save relative draft'}</button>
   {read.content.draft?<><ul>{read.content.draft.limitations.map(l=><li key={l}>{l}</li>)}</ul><ol className={styles.days}>{read.content.draft.days.map(d=><li key={d.ordinal}><h4>{zh?'第':'Day '}{d.ordinal}{zh?'天':''} · {d.destination}</h4><ul>{d.activities.map(a=><li key={a}>{a}</li>)}</ul><button type="button" disabled={!canAct} onClick={()=>setEdit({scopeKey:scopeKey!,sourceKey:sourceKey!,artifactId:read.artifactId,ordinal:d.ordinal,destination:d.destination,activities:d.activities.join('\n'),baseRevision:read.revision})}>{zh?'修改这一天':'Edit this day'}</button></li>)}</ol>
    {edit&&edit.scopeKey===scopeKey&&edit.sourceKey===sourceKey&&edit.artifactId===read.artifactId?<form onSubmit={e=>{e.preventDefault();void act('edit',{replacements:[{ordinal:edit.ordinal,destination:edit.destination.trim(),activities:edit.activities.split('\n').map(a=>a.trim()).filter(Boolean)}]});}}><label>{zh?'目的地':'Destination'}<input maxLength={80} value={edit.destination} onChange={e=>setEdit({...edit,destination:e.target.value})}/></label><label>{zh?'主题，每行一个':'One activity theme per line'}<textarea maxLength={1288} value={edit.activities} onChange={e=>setEdit({...edit,activities:e.target.value})}/></label><button type="submit" disabled={!canAct||edit.baseRevision!==read.revision}>{zh?'保存本日修改':'Save this day'}</button><button type="button" onClick={()=>setEdit(null)}>{zh?'取消修改':'Cancel edit'}</button></form>:null}
    <form onSubmit={e=>{e.preventDefault();void act('bind',{tripId,expectedTripVersion:tripVersion,startDate});}}><label>{zh?'明确开始日期后，生成同 Trip 差异':'Choose a start date to prepare the same Trip diff'}<input type="date" required value={startDate} onChange={e=>setStartDate(e.target.value)}/></label><button type="submit" disabled={!canAct||bindBlocked||read.content.draft.days.length===0}>{zh?'生成待审阅计划':'Prepare plan for review'}</button></form>
   </>:null}
   {pendingHere?<div><button type="button" disabled={!baseCanAct} onClick={()=>void act(pending.current!.action as 'choose'|'save'|'edit'|'bind',{},true)}>{zh?'重试相同请求':'Retry same request'}</button><button type="button" disabled={busy} onClick={()=>{pending.current=null;setRefresh(n=>n+1);}}>{zh?'清除未确认请求并重新读取原成果':'Clear unresolved request and reread originals'}</button></div>:null}
  </>:null}
 </section>;
}
